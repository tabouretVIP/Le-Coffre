\chapter{Chaînes de Markov en temps discret}

\section{Chaînes de Markov}

\begin{definition}
    Un processus stochastique est une collection $\{X_t\mid t\in T\}$ de v.a sur un espace $(\Omega,\mathcal{A}, P)$, indexée par un ensemble $T$. Lorsque $T= \N$, on parle de processus stochastique en temps discret et lorsque $T= \R ^{\ge0}$, on parle de processus stochastique en temps continu.
\end{definition}

On veut s'éloigner de la notion i.i.d. que l'on a pu étudier auparavant dans le but d'étudier des phenomenes plus complexes.

\begin{definition}
    Une chaine de Markov (en temps discret) est un processus stochastique $\{X_n\mid n\ge 0\}$ à valeurs dans un ensemble fini ou dénombrable $E$, qui satisfait :
    \begin{description}
        \item[Propriété de Markov : ]$$ P(X_{n+1}\mid X_0 = i_0, X_1 = i_1,\dotsc, X_n = i_n) = P(X_{n+1} = j\mid X_n = i_n) \quad \forall n \in \N, \forall j, i_0, \dotsc, i_n\in E$$ 
        \item[Homogéneité : ] $$P(X_{n+1}= j\mid X_n = 1)= P(X_1= j\mid X_0= i) \quad \forall n, \forall i,j \in E$$    
    \end{description}
\end{definition}
\begin{notation}
    Pour être clair, on établit les conventions suivantes :

    $E = \{\text{états du système\}}$

    $X_n = i$ : le système est dans l'état $i$ au moment $n$.

    $X_n = i, X_{n=1} = j$ : entre $n$ et $n+1$, le système fait une transitino de $i$ à $j$
    
    Un vecteur sera noté $\overline{v}$ et sa nature dépendra du contexte. 
    
    On utilise aussi: $\frac{1}{\infty}= 0$ et $0\cdot \infty = 0$.
\end{notation}

\begin{definition}
    La matrice de transition d'une chaîne de Markov $\{X_n\}$ est la matrice $\underline P$ de dimension $\abs{E}\times\abs{E}$ telle que $\un P_{ij} = P(X_1 = j\mid X_0 = i) = P(X_{n+1} = j\mid X_n = i)\quad \forall n$
\end{definition}

On dit aussi que $\underline P$ sera une matrice stochastique, c'est à dire que 
\begin{itemize}
    \item $P_{ij}\geq 0 \quad \forall i,j \in E$
    \item $\un P\overline{1}= \overline{ 1}$
\end{itemize}
Pour être clair :
$$(\un P\overline{1})_i= \sum_{k \in E}P_{ik}= \sum _{k\in E}P(X_1= k\mid X_0= i)= P(X_1\in E\mid X_0= i)= 1$$

\begin{definition}
    Le vecteur initial $\overline{\alpha}$ d'une CM $\{X_n\}$ est le vecteur de dim $|E|$ tel que $$\alpha_i= P(X_0= i)$$
    En fait, $\overline{\alpha}$ approxime la loi de la v.a de $X_0$ 
\end{definition}
\begin{proposition}
     Le couple $(\overline{\alpha},P)$ caractérise complètement l'évolution de la CM.
    \begin{enumerate}
        \item $$P(X_0 = i_0, X_1 = i_1,\dotsc, X_n = i_n) = \alpha_{i_0}P_{i_0i_1}P_{i_1i_2}\cdot\dots P_{i_{n-1}i_n}$$
        \item $$P(X_n= j\mid X_0= i)= (\un P^n)_{ij}$$
        \item $$P(X_n= j)= (\overline{\alpha}\un P^n)_j$$
    \end{enumerate}
\end{proposition}

Pour rappel: 
$$P(A\cap B \mid C)= P(A\mid B,C)P(B\mid C)$$

\begin{proof}
    Prouvons maintenant les trois points précédents : 
    \begin{enumerate}
        \item On a  
        \begin{align*}
           P(X_0 = \dotsc, X_n = i_n) &= P( X_1 = i_1,\dotsc, X_n = i_n\mid X_0 = i_0)P(X_0 = i_0)\\
           &=  P(X_2 = i_2,\dots, X_n = i_n\mid X_0 = i_0, X_1 = i_1)P(X_1 = i_1\mid X_0 = i_0) \alpha_{i_0}\\
           &=  P(X_3 = i_3,\dots, X_n = i_n\mid X_0 = i_0, X_1 = i_1,X_2 = i_2)P(X_2 = i_2\mid X_0 = i_0, X_1 = i_1) \un P_{01}\alpha_{i_0}\\
           &= P(X_3 = i_3,\dots, X_n = i_n\mid X_0 = i_0, X_1 = i_1,X_2 = i_2)P(X_2 = i_2\mid  X_1 = i_1) \un P_{01}\alpha_{i_0} \text{ par la propriété de Markov}
           &= \dots\\ 
        &=\alpha_{i_0}P_{i_0i_1}P_{i_1i_2}\cdot\dots P_{i_{n-1}i_n}
        \end{align*}
        \item Par induction sur $n$. Le cas de base est trivialement vérifié. Supposons que cela est vrai pour $n$ montrons le pour $n+1$
        \begin{align*}
        P(X_{n+1}=j \mid X_0= i)&=\sum_{k\in E}P(X_{n+1}=j\mid X_0=i, X_n=k)P(X_n= k\mid X_0= i)\\
        & = \sum _{k\in E}(P^n)_{ik}P_{kj}\\
        &=(P^n P)_{ij}
        \\ &= (P^{n+1})_{ij}
        \end{align*}
        \item 
        \begin{align*}
            P(X_n = j)&= \sum_{k\in E}P(X_n = j\mid X_0= k)P(X_0= k)\\ &= \sum _{k\in E} (P^{n})_{kj}\alpha_k\\
            &= (\overline{\alpha}P^n)_j
        \end{align*}
    \end{enumerate}
\end{proof}
\begin{proposition}[Équations de Chapman-Kolmogorov]
    $$P(X_{n+m} = j\mid X_0 = i) = \sum_{k\in E}(P(X_n = k\mid X_0 = i)P(X_m = j\mid X_0 = k)$$
\end{proposition}
\begin{proof}
    

On sait que $P^{n+m}= P^nP ^m$. On sait donc toujours que 
$$P(X_{n+m}= j \mid X_0 = i)= \sum_{k\in E}P(X_n= k\mid X_0= i)P(X_m= j \mid X_0= k)$$
\end{proof}

\item[Epidémies SIS :]
On suppose avoir une population de $N$ personnes elle est fermée et constante. Un virus s'y balade. On catégarise la population avec les suseptibles et les infectés et une fois infecté apres être guérris on peut redevenir malade. On est infecté au moment $n$ on est suseptible au moment $n+1$. On évalue que chaque infecté rencontre une autre personne suseptible se fait avec un probabilité $p$. Et la rencontre sus et infecté mène à la contamination avec une proba $\rho$. On note $S_n$ le nombre de sus au moment $n$ et $I_n$ et le nombre d'infectés au moment $n$. Etant donné l'état de la population nous avons pour tout $n$, $I_n= N-S_n$. Dans ce modèle, sachant $S_n$,
$$S_{n+1} \sim \operatorname{Bin}(S_n, q^{I_n})+ I_n$$

$q$ étant la probabilité qu'un suseptible fixé échape à un effecté fixé. On peut soit ne pas rencontré l'infecté ou le rencontré mais que ce contacte ne mmène pas à une contamination  c'est à dire que $$q = 1-p +\rho(1-p)= 1-\rho p$$ Donc sachant $S_n$,
$$S_{n+1} \sim \operatorname{Bin}(S_n, q^{N-S_n})+ N-S_n$$
Ainsi, $\{S_n\}$ est une chaine de Markov, avec $\overline{\alpha}= \overline{e}_{s_0}$. On trouve donc \begin{align*}
    &P_{ij}= P(X= j)\\
     &X= Y+ N - i\\
     &Y \sim \operatorname{Bin}(i, q^{N-i})
\end{align*}

\section{Propriété forte de Markov}

Soit $\{X_n\}$ une chaine de Markov, la propriété de Markov est :

$$
\forall k \in \N_0 \quad P(X_{k+1}=i_1, \dots X_{k+n}= i_n | X_0,X_1,\dots, X_k= i)= P(X_1= i_1,X_2= i_2,\dots , X_n=i_n | X_0=i)
$$
On peut se demander si on peut remplacer $k$ par \textbf{une variable aléatoire} et que cette propriété reste vraie. En effet, cela est possible si cette v.a est un temps d'arret.

\begin{definition}
    Soient $\{X_n\}$ unz CM et $T$ une v.a sur $\N \cup \{+\infty\}$. On dit que 
    $T$ est un temps d'arret pour $\{X_n\}$ si $\forall n $, l'évenement $T= n$ ne depend que de $X_0,\dots, X_n$ ( et pas du future $X_{n+1},\dots $)
\end{definition}

\begin{exemple}
    Voici deux exemples  : soit $j\in E$,
    \begin{itemize}
        \item posons $\theta_j = \operatorname{inf}\{n\geq 0, X_n = j\}$
        \item posons $L_j= \operatorname{sup}\{n\geq 0, X_n= j\}$ (n'est pas un temps d'arret)
    \end{itemize}
\end{exemple}

\begin{proposition}[Propriété de Markov forte]
Soient $\{X_n\}$ une chaîne de Markov $\sim(\alpha,P)$ et $T$ un temps d'arrêt pour $\{X_n\}$. Alors, sachant $\{T<\infty,\ X_T=i\}$, le processus $\{X_{T+m} \,:\, m\ge 0\}$ est une chaîne de Markov $\sim(e_i,P)$, indépendante de $X_0,X_1,\dots,X_T$.

Autrement dit : si $A$ est un événement qui ne dépend que de $X_0,\dots,X_T$, c'est-à-dire tel que $A\cap\{T=k\}\in\sigma(X_0,\dots,X_k)$ pour tout $k\ge0$, et si $P(A,\ T<\infty,\ X_T=i)>0$, alors pour tous $m\ge1$ et $i_1,\dots,i_m$,
\[
P\big(X_{T+1}=i_1,\dots,X_{T+m}=i_m \,\big|\, A,\ T<\infty,\ X_T=i\big)
= P\big(X_1=i_1,\dots,X_m=i_m \,\big|\, X_0=i\big).
\]
\end{proposition}

\begin{proof}
Fixons $m\ge1$ et $i_1,\dots,i_m$, et notons
\[
E:=\{X_{T+1}=i_1,\dots,X_{T+m}=i_m\},\qquad
E_k:=\{X_{k+1}=i_1,\dots,X_{k+m}=i_m\}\quad(k\ge0),
\]
ainsi que
\[
\pi:=P\big(X_1=i_1,\dots,X_m=i_m \,\big|\, X_0=i\big).
\]
On utilisera la propriété de Markov sous la forme suivante : pour tout $k\ge0$ et tout $B\in\sigma(X_0,\dots,X_k)$ tel que $P(B,\,X_k=i)>0$,
\begin{equation}\label{eq:markov}
P\big(E_k \,\big|\, B,\ X_k=i\big)=\pi .
\end{equation}

Par définition de la probabilité conditionnelle, il suffit de montrer que
\begin{equation}\label{eq:but}
P\big(E,\ A,\ T<\infty,\ X_T=i\big)=\pi\;P\big(A,\ T<\infty,\ X_T=i\big).
\end{equation}

Comme $\{T<\infty\}$ est la réunion disjointe des $\{T=k\}$, $k\ge0$, et que sur $\{T=k\}$ on a $E=E_k$ et $X_T=X_k$, on obtient
\begin{equation}\label{eq:somme}
P\big(E,\ A,\ T<\infty,\ X_T=i\big)
=\sum_{k=0}^{\infty}P\big(E,\ A,\ T=k,\ X_T=i\big)
=\sum_{k=0}^{\infty}P\big(E_k,\ A,\ T=k,\ X_k=i\big).
\end{equation}

Soit $k\ge0$. L'événement $B_k:=A\cap\{T=k\}$ appartient à $\sigma(X_0,\dots,X_k)$, par hypothèse sur $A$ et parce que $T$ est un temps d'arrêt. Si $P(B_k,\,X_k=i)=0$, alors $P(E_k,\,B_k,\,X_k=i)=0$ aussi. Sinon, d'après \eqref{eq:markov} appliquée à $B=B_k$,
\[
P\big(E_k,\ B_k,\ X_k=i\big)
=P\big(E_k\,\big|\,B_k,\ X_k=i\big)\,P\big(B_k,\ X_k=i\big)
=\pi\,P\big(B_k,\ X_k=i\big).
\]
Dans tous les cas,
\begin{equation}\label{eq:terme}
P\big(E_k,\ A,\ T=k,\ X_k=i\big)=\pi\,P\big(A,\ T=k,\ X_k=i\big).
\end{equation}

En reportant \eqref{eq:terme} dans \eqref{eq:somme}, puis en utilisant à nouveau la décomposition disjointe de $\{T<\infty\}$,
\[
P\big(E,\ A,\ T<\infty,\ X_T=i\big)
=\pi\sum_{k=0}^{\infty}P\big(A,\ T=k,\ X_k=i\big)
=\pi\,P\big(A,\ T<\infty,\ X_T=i\big),
\]
ce qui est \eqref{eq:but}. Puisque $P(A,\ T<\infty,\ X_T=i)>0$, on en déduit
\[
P\big(X_{T+1}=i_1,\dots,X_{T+m}=i_m \,\big|\, A,\ T<\infty,\ X_T=i\big)
=\pi
=P\big(X_1=i_1,\dots,X_m=i_m \,\big|\, X_0=i\big).
\]

Le membre de droite vaut $p_{ii_1}p_{i_1i_2}\cdots p_{i_{m-1}i_m}$ et ne dépend pas de $A$. En prenant $A=\Omega$, on voit que, sachant $\{T<\infty,\ X_T=i\}$, le processus $\{X_{T+m}\}_{m\ge0}$ a les lois fini-dimensionnelles d'une chaîne de Markov $\sim(e_i,P)$ ; et l'indépendance du résultat vis-à-vis de $A$ traduit précisément l'indépendance de ce processus avec $X_0,\dots,X_T$.
\end{proof}

\textbf{Application : }

Soit $(X_n)_{n\ge 0}$ une chaîne de Markov $\sim(\alpha,P)$ sur $E$ et $j\in E$ fixé. On pose
\[
T_j:=\inf\{n\ge 1 : X_n=j\}
\qquad\text{(temps de premier retour en } j\text{)},
\]
et l'on définit les temps de passages successifs en $j$ par
\[
\theta_j(1):=\inf\{n\ge 0 : X_n=j\},
\qquad
\theta_j(k+1):=\inf\{n>\theta_j(k) : X_n=j\}\quad(k\ge 1),
\]
avec la convention $\inf\emptyset=+\infty$ (et $\theta_j(k+1)=+\infty$ si $\theta_j(k)=+\infty$).
Les durées $\theta_j(k+1)-\theta_j(k)$ sont les longueurs des intervalles $I_k$ entre deux visites consécutives en $j$.

\begin{lemme}
Pour tout $k\ge 1$, $\theta_j(k)$ est un temps d'arrêt pour $(X_n)$, et $X_{\theta_j(k)}=j$ sur $\{\theta_j(k)<\infty\}$.
\end{lemme}

\begin{proof}
Par récurrence sur $k$. Pour $k=1$ et $n\ge 0$,
$\{\theta_j(1)=n\}=\{X_0\ne j,\dots,X_{n-1}\ne j,\,X_n=j\}\in\sigma(X_0,\dots,X_n)$.
Si $\theta_j(k)$ est un temps d'arrêt, alors pour $n\ge 0$,
\[
\{\theta_j(k+1)=n\}=\bigsqcup_{\ell=0}^{n-1}\{\theta_j(k)=\ell\}\cap\{X_{\ell+1}\ne j,\dots,X_{n-1}\ne j,\,X_n=j\}
\in\sigma(X_0,\dots,X_n).
\]
Enfin $X_{\theta_j(k)}=j$ sur $\{\theta_j(k)<\infty\}$ par définition de l'infimum (l'ensemble $\{n:X_n=j\}$ est un sous-ensemble de $\mathbb N$, donc l'inf est atteint lorsqu'il est fini).
\end{proof}

\begin{proposition}
Soit $k\ge 1$ tel que $P(\theta_j(k)<\infty)>0$. Alors, conditionnellement à $\{\theta_j(k)<\infty\}$, la variable $\theta_j(k+1)-\theta_j(k)$ est indépendante de $(X_0,\dots,X_{\theta_j(k)})$ et a même loi que $T_j$ sous $P(\,\cdot\mid X_0=j)$ :
\[
P\big(\theta_j(k+1)-\theta_j(k)=s \,\big|\, \theta_j(k)<\infty\big)
=P\big(T_j=s\,\big|\,X_0=j\big)\qquad\text{pour tout } s\in\mathbb N^*\cup\{+\infty\}.
\]
Plus précisément, pour tout événement $A$ déterminé par $X_0,\dots,X_{\theta_j(k)}$ tel que $P(A,\theta_j(k)<\infty)>0$,
\[
P\big(\theta_j(k+1)-\theta_j(k)=s \,\big|\, A,\ \theta_j(k)<\infty\big)
=P\big(T_j=s\,\big|\,X_0=j\big).
\]
\end{proposition}

\begin{proof}
Notons $T:=\theta_j(k)$. D'après le lemme, $T$ est un temps d'arrêt et $\{T<\infty\}=\{T<\infty,\ X_T=j\}$. On peut donc appliquer la propriété de Markov forte avec $i=j$ : posons $Y_m:=X_{T+m}$ pour $m\ge 0$ (défini sur $\{T<\infty\}$). Pour tout $A$ déterminé par $X_0,\dots,X_T$ avec $P(A,T<\infty)>0$, et tous $m\ge1$, $i_1,\dots,i_m\in E$,
\begin{equation}\label{eq:forte}
P\big(Y_1=i_1,\dots,Y_m=i_m\,\big|\,A,\ T<\infty\big)
=P\big(X_1=i_1,\dots,X_m=i_m\,\big|\,X_0=j\big).
\end{equation}

\emph{Réécriture de l'incrément.} Sur $\{T<\infty\}$, par définition,
\[
\theta_j(k+1)=\inf\{n>T : X_n=j\}=T+\inf\{m\ge 1 : X_{T+m}=j\},
\qquad\text{donc}\qquad
\theta_j(k+1)-\theta_j(k)=\inf\{m\ge 1: Y_m=j\}.
\]
C'est exactement le temps de premier retour en $j$ du processus $(Y_m)_{m\ge 0}$.

\emph{Calcul de la loi.} Soit $s\in\mathbb N^*$. L'événement $\{\inf\{m\ge1:Y_m=j\}=s\}$ s'écrit comme réunion disjointe dénombrable de cylindres :
\[
\{Y_1\ne j,\dots,Y_{s-1}\ne j,\ Y_s=j\}
=\bigsqcup_{(i_1,\dots,i_{s-1})\in(E\setminus\{j\})^{s-1}}\{Y_1=i_1,\dots,Y_{s-1}=i_{s-1},\ Y_s=j\}.
\]
Par $\sigma$-additivité et \eqref{eq:forte} appliquée terme à terme,
\begin{align*}
P\big(\theta_j(k+1)-\theta_j(k)=s\,\big|\,A,\ T<\infty\big)
&=\sum_{(i_1,\dots,i_{s-1})\in(E\setminus\{j\})^{s-1}}
  P\big(Y_1=i_1,\dots,Y_{s-1}=i_{s-1},Y_s=j\,\big|\,A,\ T<\infty\big)\\
&=\sum_{(i_1,\dots,i_{s-1})\in(E\setminus\{j\})^{s-1}}
  P\big(X_1=i_1,\dots,X_{s-1}=i_{s-1},X_s=j\,\big|\,X_0=j\big)\\
&=P\big(X_1\ne j,\dots,X_{s-1}\ne j,\ X_s=j\,\big|\,X_0=j\big)
 =P\big(T_j=s\,\big|\,X_0=j\big).
\end{align*}
Le cas $s=+\infty$ s'en déduit par passage au complémentaire :
$P(\theta_j(k+1)-\theta_j(k)=\infty\mid A,T<\infty)=1-\sum_{s\ge1}P(T_j=s\mid X_0=j)=P(T_j=\infty\mid X_0=j)$.

Le membre de droite ne dépend pas de $A$ : c'est précisément l'indépendance annoncée entre $\theta_j(k+1)-\theta_j(k)$ et $(X_0,\dots,X_{\theta_j(k)})$, sachant $\{\theta_j(k)<\infty\}$. En prenant $A=\Omega$ on obtient la formule de la loi.
\end{proof}

\begin{corollaire}[Intervalles i.i.d.]
Soit $K\ge 1$ tel que $P(\theta_j(K)<\infty)>0$. Conditionnellement à $\{\theta_j(K)<\infty\}$, les variables
$\theta_j(2)-\theta_j(1),\ \dots,\ \theta_j(K)-\theta_j(K-1)$
sont indépendantes et de même loi que $T_j$ sous $P(\,\cdot\mid X_0=j)$.
\end{corollaire}

\begin{proof}
Chaque $\theta_j(\ell+1)-\theta_j(\ell)$ est déterminé par $X_0,\dots,X_{\theta_j(\ell+1)}$ ; on applique la proposition successivement pour $k=K-1,K-2,\dots,1$ en conditionnant à chaque étape par l'événement $A$ « les incréments précédents prennent des valeurs fixées », qui est déterminé par $X_0,\dots,X_{\theta_j(k)}$.
\end{proof}

\section{Visites par un état}

Soit $(X_n)_{n\ge0}$ une chaîne de Markov sur $E$ et $j\in E$. On note
\[
V_j:=\sum_{m=0}^{\infty}\mathbf 1_{\{X_m=j\}}\in\mathbb N\cup\{+\infty\}
\]
le nombre total de visites en $j$, et
\[
T_j:=\inf\{n\ge1 : X_n=j\},\qquad
f_{ij}:=P\big(T_j<\infty\,\big|\,X_0=i\big)\quad(i\in E).
\]
On reprend les temps de passages successifs $\theta_j(k)$ de la section précédente.

\begin{theoreme}\label{thm:visites}
Soit $n\ge 0$.
\begin{enumerate}
\item Pour $i=j$ :
\[
P(V_j>n\mid X_0=j)=f_{jj}^{\,n}.
\]
Par conséquent,
\[
P(V_j=n\mid X_0=j)=f_{jj}^{\,n-1}(1-f_{jj})\qquad(n\ge1).
\]
\item Pour $i\neq j$ :
\[
P(V_j>n\mid X_0=i)=f_{ij}\,f_{jj}^{\,n}.
\]
Par conséquent,
\[
P(V_j=n\mid X_0=i)=
\begin{cases}
1-f_{ij} & \text{si } n=0,\\[2pt]
f_{ij}\,f_{jj}^{\,n-1}(1-f_{jj}) & \text{si } n\ge1.
\end{cases}
\]
\end{enumerate}
\end{theoreme}

\begin{remarque}
\begin{itemize}
\item On rappelle que $Y\sim\mathcal G(p)$ (loi géométrique sur $\mathbb N^*$) si
\[
P(Y=n)=(1-p)^{n-1}p\qquad(n\ge1).
\]
Le théorème dit donc que, si $f_{jj}<1$, la loi de $V_j$ sous $P(\,\cdot\mid X_0=j)$ est $\mathcal G(1-f_{jj})$.
\item La loi géométrique est la seule loi sur $\mathbb N^*$ ayant la propriété d'absence de mémoire :
\[
P(Y>n+s\mid Y>n)=P(Y>s)\qquad(n,s\ge0).
\]
C'est exactement ce que traduit la propriété de Markov forte : à chaque retour en $j$, la chaîne « repart à neuf ».
\item Si $f_{jj}=1$, on a $P(V_j>n\mid X_0=i)=f_{ij}$ pour tout $n$. Donc $P(V_j=\infty\mid X_0=j)=1$ et $P(V_j=\infty\mid X_0=i)=f_{ij}$.
\item Il reste à calculer les $f_{ij}$.
\end{itemize}
\end{remarque}

\begin{proof}
\emph{Cas $i=j$.} Sous $X_0=j$, on a $\theta_j(1)=0$. Le nombre de visites en $j$ est donc au moins $n+1$ si et seulement si le $(n+1)$-ième passage a lieu :
\[
\{V_j>n\}=\{\theta_j(n+1)<\infty\}\qquad\text{sur }\{X_0=j\}.
\]
Montrons par récurrence sur $n$ que $P(V_j>n\mid X_0=j)=f_{jj}^{\,n}$.

Pour $n=0$, on a $P(V_j>0\mid X_0=j)=1=f_{jj}^{\,0}$, puisque $X_0=j$.

Soit $n\ge1$ ; supposons le résultat vrai au rang $n-1$. Comme $\theta_j(n+1)<\infty$ implique $\theta_j(n)<\infty$,
\begin{align*}
P(V_j>n\mid X_0=j)
&=P\big(\theta_j(n)<\infty,\ \theta_j(n+1)-\theta_j(n)<\infty\,\big|\,X_0=j\big)\\
&=P\big(\theta_j(n+1)-\theta_j(n)<\infty\,\big|\,\theta_j(n)<\infty,\ X_0=j\big)\,
  P\big(\theta_j(n)<\infty\,\big|\,X_0=j\big).
\end{align*}
Si $P(\theta_j(n)<\infty\mid X_0=j)=0$, les deux membres sont nuls et il n'y a rien à montrer. Sinon, on raisonne comme suit.
\begin{itemize}
\item L'événement $A=\{X_0=j\}$ est déterminé par $X_0,\dots,X_{\theta_j(n)}$. La proposition sur les intervalles entre visites donne donc
\[
P\big(\theta_j(n+1)-\theta_j(n)<\infty\,\big|\,\theta_j(n)<\infty,\ X_0=j\big)
=P(T_j<\infty\mid X_0=j)=f_{jj}.
\]
\item Par hypothèse de récurrence,
\[
P\big(\theta_j(n)<\infty\,\big|\,X_0=j\big)=P(V_j>n-1\mid X_0=j)=f_{jj}^{\,n-1}.
\]
\end{itemize}
Ainsi $P(V_j>n\mid X_0=j)=f_{jj}\cdot f_{jj}^{\,n-1}=f_{jj}^{\,n}$, ce qui achève la récurrence.

\emph{Cas $i\neq j$.} On décompose selon que la chaîne atteint $j$ ou non :
\begin{align*}
P(V_j>n\mid X_0=i)
={}&P\big(V_j>n\,\big|\,X_0=i,\ T_j<\infty\big)\,P(T_j<\infty\mid X_0=i)\\
&+P\big(V_j>n\,\big|\,X_0=i,\ T_j=\infty\big)\,P(T_j=\infty\mid X_0=i),
\end{align*}
avec la convention qu'un terme est nul si la probabilité de l'événement conditionnant est nulle.
\begin{itemize}
\item \emph{Second terme.} Sur $\{X_0=i,\ T_j=\infty\}$, la chaîne ne visite jamais $j$, donc $V_j=0$. Le second terme est nul.
\item \emph{Premier terme.} Comme $X_0=i\neq j$, la chaîne ne visite pas $j$ avant $T_j$. Sur $\{T_j<\infty\}$, on a donc
\[
V_j=\sum_{m\ge0}\mathbf 1_{\{X_{T_j+m}=j\}}.
\]
Autrement dit, $V_j$ est le nombre de visites en $j$ du processus $(X_{T_j+m})_{m\ge0}$.

Or $T_j$ est un temps d'arrêt, $X_{T_j}=j$ sur $\{T_j<\infty\}$, et $\{X_0=i\}$ est déterminé par $X_0,\dots,X_{T_j}$. Par la propriété de Markov forte, sous $P(\,\cdot\mid X_0=i,\ T_j<\infty)$, le processus $(X_{T_j+m})_{m\ge0}$ a la loi d'une chaîne $(\delta_j,P)$.

L'événement $\{V_j>n\}$ se décompose en réunion dénombrable de cylindres. Par $\sigma$-additivité, comme dans la section précédente, on obtient
\[
P\big(V_j>n\,\big|\,X_0=i,\ T_j<\infty\big)=P(V_j>n\mid X_0=j)=f_{jj}^{\,n}.
\]
\end{itemize}
D'où
\[
P(V_j>n\mid X_0=i)=f_{ij}\,f_{jj}^{\,n}.
\]

\emph{Lois ponctuelles.} Pour $n\ge1$, on a $\{V_j=n\}=\{V_j>n-1\}\setminus\{V_j>n\}$. Par conséquent,
\[
P(V_j=n\mid X_0=i)=f_{ij}f_{jj}^{\,n-1}-f_{ij}f_{jj}^{\,n}=f_{ij}\,f_{jj}^{\,n-1}(1-f_{jj}),
\]
et, pour $i\neq j$,
\[
P(V_j=0\mid X_0=i)=1-P(V_j>0\mid X_0=i)=1-f_{ij}.
\]
Le cas $i=j$ s'obtient de la même façon, en remplaçant $f_{ij}$ par $1$.
\end{proof}

\begin{definition}
Pour $i,j\in E$, on note
\[
R_{ij}:=\mathbb E\big[V_j\,\big|\,X_0=i\big]\in[0,+\infty]
\]
le nombre moyen de visites en $j$ partant de $i$.
\end{definition}

\begin{proposition}
Avec la convention $\dfrac{1}{0}=+\infty$ (et $R_{ij}=0$ si $f_{ij}=0$),
\[
R_{ij}=
\begin{cases}
\dfrac{1}{1-f_{jj}} & \text{si } i=j,\\[10pt]
\dfrac{f_{ij}}{1-f_{jj}} & \text{si } i\neq j.
\end{cases}
\]
En particulier, $R_{jj}=+\infty$ si et seulement si $f_{jj}=1$.
\end{proposition}

\begin{proof}
$V_j$ est à valeurs dans $\mathbb N\cup\{+\infty\}$. On peut donc écrire
\[
V_j=\sum_{n\ge0}\mathbf 1_{\{V_j>n\}}.
\]
Par convergence monotone (Fubini--Tonelli), puis par le théorème~\ref{thm:visites}, on obtient pour $i\neq j$
\[
R_{ij}=\sum_{n=0}^{\infty}P(V_j>n\mid X_0=i)
=\sum_{n=0}^{\infty}f_{ij}\,f_{jj}^{\,n}
=\frac{f_{ij}}{1-f_{jj}}.
\]
La somme géométrique vaut $+\infty$ si $f_{jj}=1$ et $f_{ij}>0$ ; elle vaut $0$ si $f_{ij}=0$.

Pour $i=j$, le même calcul avec $P(V_j>n\mid X_0=j)=f_{jj}^{\,n}$ donne
\[
R_{jj}=\frac{1}{1-f_{jj}}.
\]
\end{proof}

\begin{proposition}
La matrice $R=(R_{ij})_{i,j\in E}$ vérifie
\[
R=\sum_{n=0}^{\infty}P^n,
\qquad\text{c'est-à-dire}\qquad
R_{ij}=\sum_{n=0}^{\infty}(P^n)_{ij}\in[0,+\infty]\quad\text{pour tous } i,j\in E.
\]
\end{proposition}

\begin{proof}
Soient $i,j\in E$. Par définition de $V_j$,
\[
R_{ij}=E\big[V_j\,\big|\,X_0=i\big]
=E\Big[\sum_{n=0}^{\infty}\mathbf 1_{\{X_n=j\}}\,\Big|\,X_0=i\Big].
\]
Les variables $\mathbf 1_{\{X_n=j\}}$ sont positives. Le théorème de Fubini--Tonelli (ou de convergence monotone) permet donc d'échanger somme et espérance, y compris lorsque la somme vaut $+\infty$ :
\[
R_{ij}=\sum_{n=0}^{\infty}E\big[\mathbf 1_{\{X_n=j\}}\,\big|\,X_0=i\big]
=\sum_{n=0}^{\infty}P\big(X_n=j\,\big|\,X_0=i\big).
\]
Enfin, $P(X_n=j\mid X_0=i)=(P^n)_{ij}$ (probabilité de transition en $n$ pas), d'où
\[
R_{ij}=\sum_{n=0}^{\infty}(P^n)_{ij}.
\]
\end{proof}
